# ============================================================
# requirements.txt для Globalizer

# Покрывает: задачу iOptProblemSimple + визуализацию (lib/plotters)
#            + Python-интерфейс PYGlobalizer.
# ============================================================

# ---------- Ядро: numpy (используется везде) ----------
# numpy 2.x требует Python >=3.10; для 3.8/3.9 остаёмся на 1.24.x.
numpy>=1.24,<2.0 ; python_version < "3.10"
numpy>=1.26,<3.0 ; python_version >= "3.10"

# ---------- Задача iOptProblemSimple ----------
# Globalizer_problem.py: requests, tqdm
requests>=2.28,<3
tqdm>=4.64,<5

# ---------- Визуализация (lib/plotters) ----------
# painters.py / plotters.py: matplotlib, scipy, scikit-learn.
# Диапазоны подобраны так, чтобы pip сам взял версию под текущий Python:
#   * matplotlib 3.8+ тянет contourpy автоматически (не пиним его!);
#   * scipy / scikit-learn имеют разные min-Python — маркеры ниже.

matplotlib>=3.6,<4

# scipy: 1.11+ требует Python >=3.9; 1.13+ — >=3.9; на 3.8 берём 1.10.x
scipy>=1.10,<1.11 ; python_version < "3.9"
scipy>=1.10,<2    ; python_version >= "3.9"

# scikit-learn: 1.4+ требует Python >=3.9; на 3.8 берём 1.3.x
scikit-learn>=1.2,<1.4 ; python_version < "3.9"
scikit-learn>=1.2,<2   ; python_version >= "3.9"

# ---------- Python-интерфейс PYGlobalizer (pybind11) ----------
# Нужен ТОЛЬКО при сборке PYGlobalizer (флаг -DPYGlobalizer=ON).
# Если pybind11 подключён как submodule/через CMake — строку можно закомментировать.
pybind11>=2.11,<3
